3Define study 2 analytical population

# Women reporting at least one of:
# V763A = STI in last 12 months
# V763B = genital sore/ulcer
# V763C = genital discharge
# V763D = genital itching
 
study2_eligible_v770 <- with(
  ndhs,
  V763A == 1 |
    V763B == 1 |
    V763C == 1 |
    V763D == 1
)
 
 
# Check eligibility
table(

  useNA = "ifany"

15Sources of advice/treatment

source_vars <- grep(
  "^V770[A-Z]",
  names(study2_data),
  value = TRUE
)
 
 
source_summary <- data.frame(
  Variable = source_vars,
  
  Source = sapply(
    study2_data[source_vars],
    function(x) attr(x, "label")
  ),
  
  n = sapply(
    study2_data[source_vars],
    function(x) sum(x == 1, na.rm = TRUE)
  )
)
 
 
# Unweighted percentage among those who sought care
source_summary$Percent_unweighted_among_seekers <-
  source_summary$n /
  sum(study2_data$V770 == 1) *
  100
 
 
source_summary

16Survey-weighted source-of-care estimates

seekers_design <- subset(
  study2_design,
  V770 == 1
)
 
 
source_weighted <- lapply(
  source_vars,
  function(v) {
    
    f <- as.formula(
      paste0("~", v)
    )
    
    est <- svymean(
      f,
      seekers_design,
      na.rm = TRUE
    )
    
    ci <- confint(est)
    
    data.frame(
      Variable = v,
      n = sum(
        study2_data[study2_data$V770 == 1, v] == 1,
        na.rm = TRUE
      ),
      Weighted_percent =
        as.numeric(coef(est)) * 100,
      Lower_CI =
        as.numeric(ci[, 1]) * 100,
      Upper_CI =
        as.numeric(ci[, 2]) * 100
    )
  }
)
 
 
source_weighted <- bind_rows(
  source_weighted
)
 
 
source_weighted
